\begin{frame}{Why regulated documents are a data-management problem}
  \begin{itemize}
    \item RAG grounds LLMs in private, factual, regulated collections: sustainability reports (EU CSRD/ESRS), financial filings, legal and clinical records
    \item Practitioners must choose chunking, indexing, embedding models and LLMs
    \item Hard question: is the end-to-end system \textbf{faithful, attributable, auditable, reproducible} at scale?
    \item Scale and heterogeneity (PDF, HTML, XML; EN/IT) make this a database and data-engineering problem
  \end{itemize}
\end{frame}

\begin{frame}{Part 1 --- RAG paradigms and trustworthiness}
  \begin{itemize}
    \item Paradigms: naive $\rightarrow$ advanced (rerank, rewrite, decompose) $\rightarrow$ modular RAG
    \item Retrieval--generation coupling for knowledge-intensive tasks
    \item Trustworthiness desiderata:
    \begin{itemize}
      \item faithfulness and provenance/attribution
      \item verifiability and temporal consistency
      \item auditability and explainability
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Part 2 --- Vector data management and filtered retrieval}
  \begin{itemize}
    \item Vector indexes: HNSW/IVF, disk-resident billion-scale ANN (DiskANN), GPU search (FAISS)
    \item Vector DBMS landscape: Milvus, Manu, and others
    \item \textbf{Filtered and hybrid retrieval:} predicates on company, report year, GRI/ESRS standard
    \item Interplay of filtering, ANN indexing, hybrid scoring, and index freshness
    \item Systems: partitioning, replication, algorithm--system co-design for RAG serving
  \end{itemize}
\end{frame}

% 7
\begin{frame}{Hybrid document--database analytics}
  \begin{itemize}
    \item Route each question to:
    \begin{itemize}
      \item evidence-grounded retrieval over report narratives, or
      \item schema-grounded Text-to-SQL over extracted ESG tables
    \end{itemize}
    \item Enforce read-only SQL execution
    \item Keep both the SQL and the source passages as provenance
  \end{itemize}
\end{frame}

% 8
\begin{frame}{Part 3 --- Metrics and benchmark design}
  \begin{itemize}
    \item Retrieval: Recall/Precision@k, NDCG
    \item Generation: faithfulness, answer relevancy, factual correctness
    \item Benchmarks: BEIR, MTEB; RAGAS for automated evaluation
    \item Reproducible ANN evaluation (ANN-Benchmarks)
    \item \textbf{LLM-as-judge reliability:} nondeterminism, judge bias, calibration on a human-validated sample
    \item Test-set generation without human labels
  \end{itemize}
\end{frame}

% 9
\begin{frame}{Part 3 --- Data lineage and systems metrics}
  \begin{itemize}
    \item Lineage as queryable records: answer claims, source passages, corpus version, retrieval configuration, prompts/models, executed SQL
    \item Systems metrics next to answer quality:
    \begin{itemize}
      \item latency, index-build time, memory footprint
      \item tokens and cost per query
    \end{itemize}
    \item Output: quality--cost--latency trade-offs for each configuration
  \end{itemize}
\end{frame}
