\begin{frame}{ESG/regulatory analytics as a data-management problem}
  \begin{itemize}
    \item ESG and regulated disclosures are heterogeneous, multilingual, and frequently revised (narratives, tables, digital artifacts)
    \item Core data-management concerns: corpus acquisition/versioning, parsing and structuring, metadata/schema/indicator mapping, data quality and comparability, provenance, temporal consistency, auditability
    \item Reporting landscape: CSRD/ESRS, GRI, ISSB/IFRS, TCFD; growing machine-readable reporting (e.g. XBRL taxonomies)
    \item Retrieval-Augmented Generation (RAG) and hybrid analytics ground LLMs in evidence, but require trustworthy, reproducible data pipelines and evaluation
  \end{itemize}
\end{frame}

% 3
\begin{frame}{Greenwashing and evidence-grounded claim verification}
  \begin{itemize}
    \item Risks: selective disclosure, vague/unsupported claims, inconsistencies between narratives and evidence
    \item Need traceable evidence, not opaque scores: link claims to passages, extracted tables/rows, and provenance
    \item Evaluation perspective: supported / contradicted / unverifiable claims, temporal consistency across reporting periods
    \item Connects to broader NLP/data work on climate/ESG claim analysis and verification (e.g. disclosure analysis, claim detection, evidence-based verification)
  \end{itemize}
\end{frame}

\begin{frame}{P1 --- Foundations, ESG data management, and RAG paradigms}
  \begin{itemize}
    \item Standards/frameworks and interoperability: CSRD/ESRS, GRI, ISSB/IFRS, TCFD; digital reporting (XBRL taxonomies)
    \item Corpus acquisition/versioning, metadata, schema/indicator mapping, quality and comparability
    \item RAG paradigms: naive $\rightarrow$ advanced (rerank, rewrite, decompose) $\rightarrow$ modular/self-reflective
    \item Provenance and governance requirements for regulated analytics
  \end{itemize}
\end{frame}

% 6
\begin{frame}{P2 --- Document/data pipelines, indexing, and hybrid analytics}
  \begin{itemize}
    \item Parse/structure heterogeneous formats (PDF/HTML/XML), layout/table extraction, document understanding
    \item Chunking and embeddings; vector indexes (HNSW/IVF/DiskANN) and vector DBMS landscape
    \item Metadata-aware/filtered retrieval, hybrid search (BM25+dense), late interaction, re-ranking
    \item Hybrid document--database analytics: retrieval over narratives + schema-grounded Text-to-SQL over extracted tables, with provenance (passages, rows, SQL)
    \item 3h depth: recall--latency--memory--build-time trade-offs, filtering modes, freshness/update cost
  \end{itemize}
\end{frame}

% 7
\begin{frame}{P3--P4 --- Evaluation, benchmarking, provenance, and claim verification}
  \begin{itemize}
    \item Metrics/benchmarks: retrieval (Recall/Precision@k, nDCG), generation (faithfulness, relevancy, factuality), end-to-end; BEIR/MTEB, RAGAS and LLM-as-judge reliability
    \item Reproducible systems evaluation: controlled comparisons, ANN benchmarking, quality--cost frontiers (latency, index build, memory, tokens/cost)
    \item Lineage as first-class output: claims, passages, corpus versions, configs, prompts/models, executed SQL
    \item Evidence-grounded claim verification/greenwashing analysis: support/contradiction/unverifiability, temporal consistency, selective disclosure
    \item Illustrative worked example: multilingual ESG RAG/benchmarking over a large report corpus (methodology-focused)
  \end{itemize}
\end{frame}

% 8
\begin{frame}{P5 --- Open challenges}
  \begin{itemize}
    \item Robust claim substantiation and auditable greenwashing-signal analysis with traceable evidence
    \item Temporal consistency and evolving standards/disclosures; interoperability (CSRD/ESRS, GRI, ISSB, XBRL)
    \item Data quality, comparability, governance for fragmented ESG data; knowledge-graph/ontology-assisted integration
    \item Agentic/multi-step retrieval and verification; explainability, audit trails, human-in-the-loop
    \item Cost/performance co-design; shareable benchmarks, datasets, and lineage artifacts
  \end{itemize}
\end{frame}

% 9
\begin{frame}{P6--P7 --- Hands-on lab (3 h) and debrief}
  \begin{itemize}
    \item Prepared RAG pipeline over regulated-document corpus (Python, Docker vector store, quantized open-source LLMs/embeddings via Ollama)
    \item Metadata-aware retrieval and optional schema-grounded Text-to-SQL; inspect evidence, answers, provenance, and evaluation/claim-support signals
    \item \textbf{Local path (optional):} Linux, macOS, or Windows 11 with WSL2; Python 3.10+, Docker, Ollama; $\geq$8\,GB RAM recommended
    \item \textbf{Fallback/no-setup:} pre-pulled images and cached models; pre-generated outputs and pre-computed evaluation results; follow-along mode for all attendees
    \item Debrief (15 min): behaviors, lineage/attribution, trade-offs, Q\&A
  \end{itemize}
\end{frame}
